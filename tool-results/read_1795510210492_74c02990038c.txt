    96→  const [scanning, setScanning] = useState(false);
    97→  const [bankIdx, setBankIdx] = useState(0);
    98→  const [patchTrayOpen, setPatchTrayOpen] = useState(true);
    99→  const [fastChRms, setFastChRms] = useState<number[]>([]);
   100→  const [fastChPeak, setFastChPeak] = useState<number[]>([]);
   101→  const [fastPostDb, setFastPostDb] = useState(-96);
   102→
   103→  // Throttled ~12.5fps live hardware level meter sampling (pauses while hidden & skips unchanged silence)
   104→  useEffect(() => {
   105→    if (!engineRunning) {
   106→      setFastChRms([]);
   107→      setFastChPeak([]);
   108→      setFastPostDb(-96);
   109→      return;
   110→    }

   112→    let cancelled = false;
   113→    let prevRms: number[] = [];
   114→    let prevPeak: number[] = [];
   115→    let prevPost = -96;
   116→    const qHalf = (v: number) => (v <= -60 ? -96 : Math.round(v * 2) / 2);
   117→    const sameArr = (a: number[], b: number[]) =>
   118→      a.length === b.length && a.every((val, i) => val === b[i]);
   119→
   120→    const tick = () => {
   121→      timerId = null;
   122→      if (document.hidden) {
   123→        return;
   124→      }
   125→      const snap = liveAudioEngine.sampleFastLevels();
   126→      if (snap.isRunning) {
   127→        const nextRms = snap.channelRmsDb.map(qHalf);
   128→        const nextPeak = snap.channelPeakDb.map(qHalf);
   129→        const nextPost = qHalf(snap.postRmsDb);
   130→        if (!sameArr(nextRms, prevRms)) {
   131→          prevRms = nextRms;
   132→          setFastChRms(nextRms);
   133→        }
   134→        if (!sameArr(nextPeak, prevPeak)) {
   135→          prevPeak = nextPeak;
   136→          setFastChPeak(nextPeak);
   137→        }
   138→        if (nextPost !== prevPost) {
   139→          prevPost = nextPost;
   140→          setFastPostDb(nextPost);
   141→        }
   142→      }
   143→      if (!cancelled) timerId = window.setTimeout(tick, 80);
   144→    };
   145→
   146→    const onVis = () => {
   147→      if (!document.hidden && timerId === null) {
   148→        tick();
   149→      }
   150→    };
   151→
   152→    if (!document.hidden) tick();
   153→    document.addEventListener('visibilitychange', onVis);
   154→    return () => {
   155→      cancelled = true;
   156→      if (timerId !== null) window.clearTimeout(timerId);
   157→      document.removeEventListener('visibilitychange', onVis);
   158→    };
   159→  }, [engineRunning]);

   161→  const totalBanks = Math.max(1, Math.ceil(channels.length / 8));
   162→  const safeBankIdx = Math.min(bankIdx, totalBanks - 1);
   163→  const bankStart = safeBankIdx * 8;
   164→  const bankChannels = channels.slice(bankStart, bankStart + 8);
   165→  const selectedChannel = channels.find((c) => c.id === selectedChannelId) ?? channels[0];
   166→
   167→  const handleScan = async () => {
   168→    setScanning(true);
   169→    try {
   170→      if (onScanAudioDevices) {
   171→        await onScanAudioDevices(true);
   172→      }
   173→    } finally {
   174→      setScanning(false);
   175→    }
   176→  };
   177→
   178→  const classifyInput = (label: string) => {
   179→    const l = label.toLowerCase();
   180→    if (l.includes('desktop audio') || l.startsWith('app:')) {
   181→      return { kind: 'Desktop Audio', dot: 'bg-cyan-400', badge: 'text-cyan-300 border-cyan-500/35 bg-cyan-950/40' };
   182→    }
   183→    if (l.includes('cable') || l.includes('virtual') || l.includes('voicemeeter')) {
   184→      return { kind: 'Virtual', dot: 'bg-emerald-400', badge: 'text-emerald-300 border-emerald-500/35 bg-emerald-950/40' };
   185→    }
   186→    if (
   187→      l.includes('usb') ||
   188→      l.includes('focusrite') ||
   189→      l.includes('behringer') ||
   190→      l.includes('yamaha') ||
   191→      l.includes('scarlett') ||
   192→      l.includes('mg')
   193→    ) {
   194→      return { kind: 'USB Audio', dot: 'bg-sky-400', badge: 'text-sky-300 border-sky-500/35 bg-sky-950/40' };
   195→    }
   196→    if (l.includes('line') || l.includes('aux')) {
   197→      return { kind: 'Line In', dot: 'bg-violet-400', badge: 'text-violet-300 border-violet-500/35 bg-violet-950/40' };
   198→    }
   199→    return { kind: 'Mic In', dot: 'bg-amber-400', badge: 'text-amber-300 border-amber-500/35 bg-amber-950/40' };
   200→  };
   201→
   202→  const classifyOutput = (label: string) => {
   203→    const l = label.toLowerCase();
   204→    if (l.includes('cable') || l.includes('virtual') || l.includes('voicemeeter') || l.includes('obs')) {
   205→      return { kind: 'Virtual / OBS Bus', color: 'text-violet-300 bg-violet-950/40 border-violet-500/35', icon: Radio };
   206→    }
   207→    if (l.includes('headphone') || l.includes('earphone') || l.includes('bluetooth') || l.includes('buds')) {
   208→      return { kind: 'Headphones / Cue', color: 'text-sky-300 bg-sky-950/40 border-sky-500/35', icon: Headphones };
   209→    }
   210→    if (
   211→      l.includes('usb') ||
   212→      l.includes('interface') ||
   213→      l.includes('scarlett') ||
   214→      l.includes('behringer') ||
   215→      l.includes('yamaha')
   216→    ) {
   217→      return { kind: 'USB Audio Out', color: 'text-amber-300 bg-amber-950/40 border-amber-500/35', icon: Speaker };
   218→    }
   219→    return { kind: 'Speaker Output', color: 'text-emerald-300 bg-emerald-950/40 border-emerald-500/35', icon: Speaker };
   220→  };
   221→
   222→  const meterWidth = Math.max(0, Math.min(100, (((engineRunning ? fastPostDb : telemetry.postRmsDb) + 60) / 60) * 100));
   223→
   224→  if (collapsed) {
   225→    return (
   226→      <aside
   227→        aria-label="Collapsed Input & Output Panel"
   228→        className="w-10 shrink-0 h-full hw-panel rounded-lg flex flex-col items-center justify-between py-2.5 select-none"
   229→      >
   230→        <button
   231→          type="button"
   232→          onClick={() => setCollapsed(false)}
   233→          title="Expand Inputs / Outputs Panel"
   234→          className="p-1.5 rounded bg-[#0E141A] border border-[#2A3646] text-teal-300 hover:text-white cursor-pointer"
   235→        >
   236→          <ChevronRight className="w-3.5 h-3.5" />
   237→        </button>
   238→
   239→        <div className="flex flex-col items-center gap-3">
   240→          <button
   241→            type="button"
   242→            onClick={() => {
   243→              setActiveTab('inputs');
   244→              setCollapsed(false);
   245→            }}
   246→            title={`Inputs (${channels.length} channels)`}
   247→            className={`p-1.5 rounded border cursor-pointer ${
   248→              activeTab === 'inputs'
   249→                ? 'bg-teal-500/20 border-teal-400/60 text-teal-300'
   250→                : 'border-transparent text-slate-400 hover:text-white'
   251→            }`}
   252→          >
   253→            <Mic className="w-3.5 h-3.5" />
   254→          </button>
   255→          <button
   256→            type="button"
   257→            onClick={() => {
   258→              setActiveTab('outputs');
   259→              setCollapsed(false);
   260→            }}
   261→            title={`Outputs (${outputDevices.length} endpoints)`}
   262→            className={`p-1.5 rounded border cursor-pointer ${
   263→              activeTab === 'outputs'
   264→                ? 'bg-emerald-500/20 border-emerald-400/60 text-emerald-300'
   265→                : 'border-transparent text-slate-400 hover:text-white'
   266→            }`}
   267→          >
   268→            <Volume2 className="w-3.5 h-3.5" />
   269→          </button>
   270→        </div>
   271→
   272→        <span
   273→          className={`w-2 h-2 rounded-full ${
   274→            engineRunning ? 'bg-emerald-400 shadow-[0_0_6px_rgba(74,222,128,0.85)]' : 'bg-slate-600'
   275→          }`}
   276→        />
   277→      </aside>
   278→    );
   279→  }
   280→
   281→  return (
   282→    <aside
   283→      aria-label="Live Console Inputs & Outputs Rack"
   284→      className="w-[214px] 2xl:w-[248px] shrink-0 h-full hw-panel rounded-lg flex flex-col overflow-hidden select-none"
   285→    >

   288→        <div className="flex items-center gap-0.5 2xl:gap-1 flex-1 min-w-0 font-mono text-[8.5px] 2xl:text-[9.5px] font-bold">
   289→          <button
   290→            type="button"
   291→            onClick={() => setActiveTab('inputs')}
   292→            className={`flex-1 min-w-0 h-6 px-1 2xl:px-1.5 rounded flex items-center justify-center gap-0.5 2xl:gap-1 whitespace-nowrap cursor-pointer ${
   293→              activeTab === 'inputs' ? 'hw-top-btn-active' : 'hw-top-btn'
   294→            }`}
   295→          >
   296→            <Mic className="w-2.5 h-2.5 shrink-0" />
   297→            <span>INPUTS</span>
   298→          </button>
   299→          <button
   300→            type="button"
   301→            onClick={() => setActiveTab('outputs')}
   302→            className={`flex-1 min-w-0 h-6 px-1 2xl:px-1.5 rounded flex items-center justify-center gap-0.5 2xl:gap-1 whitespace-nowrap cursor-pointer ${
   303→              activeTab === 'outputs' ? 'hw-top-btn-active' : 'hw-top-btn'
   304→            }`}
   305→          >
   306→            <Volume2 className="w-2.5 h-2.5 shrink-0" />
   307→            <span>OUT ({outputDevices.length})</span>
   308→          </button>
   309→        </div>
   310→
   311→        <div className="flex items-center gap-0.5 2xl:gap-1 shrink-0">
   312→          <button
   313→            type="button"
   314→            disabled={scanning}
   315→            onClick={() => void handleScan()}
   316→            title="Re-Scan Live Audio Hardware & Virtual Ports"
   317→            className="p-1 rounded hw-top-btn cursor-pointer disabled:opacity-40"
   318→          >
   319→            <RefreshCw className={`w-3 h-3 ${scanning ? 'animate-spin text-teal-300' : ''}`} />
   320→          </button>
   321→          {activeTab === 'inputs' && onAddChannel && (
   322→            <button
   323→              type="button"
   324→              disabled={locked}
   325→              onClick={() => {
   326→                onAddChannel();
   327→                setBankIdx(Math.floor(channels.length / 8));
   328→              }}
   329→              title="Add New Standalone Stereo Input Strip"
   330→              className="p-1 rounded hw-top-btn cursor-pointer disabled:opacity-40"
   331→            >
   332→              <Plus className="w-3.5 h-3.5" />
   333→            </button>
   334→          )}
   335→          <button
   336→            type="button"
   337→            onClick={() => setCollapsed(true)}
   338→            title="Collapse Left Panel"
   339→            className="p-1 rounded hw-top-btn cursor-pointer"
   340→          >
   341→            <ChevronLeft className="w-3.5 h-3.5" />
   342→          </button>
   343→        </div>
   344→      </div>
   345→
   346→      {/* =========================================================================
   347→          TAB 1: STANDALONE STEREO INPUT CHANNEL RACK + HARDWARE PATCH TRAY
   348→          ========================================================================= */}

   350→        <div className="flex-1 min-h-0 flex flex-col overflow-hidden">

   361→                >
   362→                  CH {i * 8 + 1}–{Math.min(channels.length, (i + 1) * 8)}
   363→                </button>
   364→              ))}
   365→            </div>
   366→          )}
   367→
   368→          {/* Stacked 8 Machined Input Channel Cards */}
   369→          <div className="flex-1 min-h-0 overflow-y-auto p-1.5 flex flex-col gap-1.5">

   372→              const isSelected = ch.id === selectedChannel?.id;
   373→              const isBridgeSlot =

   375→              const isAssigned = Boolean(ch.hardwareInputId) || isBridgeSlot;
   376→              const rmsDb =
   377→                isAssigned && engineRunning && !ch.mute
   378→                  ? fastChRms[globalIdx] ?? telemetry.channelLevels?.[globalIdx]?.rmsDb ?? ch.levelDb

   387→                isAssigned && engineRunning && !ch.mute ? Math.max(0, Math.min(1, (rmsDb + 54) / 54)) : 0;

   389→                isAssigned && engineRunning && !ch.mute && rmsDb > -53 ? Math.max(1, Math.round(normMeter * 10)) : 0;
   390→              const faderPct = isAssigned ? Math.max(0, Math.min(100, ((ch.faderDb + 48) / 58) * 100)) : 0;

   392→              return (
   393→                <div
   394→                  key={ch.id}
   395→                  onClick={() => onSelectChannel(ch.id)}
   396→                  className={`rounded-md border px-1.5 py-1 flex items-center gap-1.5 cursor-pointer transition-all shrink-0 ${
   397→                    !isAssigned
   398→                      ? isSelected
   399→                        ? 'border-dashed border-[#4B5D72] bg-[#121922] opacity-85'
   400→                        : 'border-dashed border-[#2A3544] bg-[#0D1219]/75 opacity-60 hover:opacity-85 hover:border-[#3D4D61]'
   401→                      : isHot
   402→                        ? 'border-amber-400 bg-gradient-to-b from-[#2A2015] to-[#1A130D] shadow-[inset_0_1px_0_rgba(255,255,255,0.14),0_0_10px_rgba(245,158,11,0.28)]'
   403→                        : isSelected
   404→                          ? 'border-[#2DD4BF] ring-1 ring-[#2DD4BF]/35 bg-gradient-to-b from-[#17282D] to-[#111D22] shadow-[inset_0_1px_0_rgba(255,255,255,0.18),0_0_12px_rgba(45,212,191,0.3)]'
   405→                          : 'border-[#4B5D72] bg-gradient-to-b from-[#1E2732] to-[#131921] shadow-[inset_0_1px_0_rgba(255,255,255,0.14)] hover:border-[#687F99]'
   406→                  }`}
   407→                >
   408→                  {/* Column 1: Channel Number & Peak/Limit Indicator */}
   409→                  <div className="w-4 shrink-0 flex flex-col items-center justify-between self-stretch py-0.5 font-mono">
   410→                    <span
   411→                      className={`text-[10px] font-bold ${
   412→                        !isAssigned ? 'text-slate-600' : isSelected ? 'text-[#5EEAD4]' : 'text-slate-300'
   413→                      }`}
   414→                    >
   415→                      {ch.index}
   416→                    </span>
   417→                    {!isAssigned ? (
   418→                      <span className="w-1.5 h-1.5 rounded-full bg-[#1B232E]" />
   419→                    ) : isHot ? (
   420→                      <AlertTriangle className="w-3 h-3 text-amber-400 animate-pulse" />
   421→                    ) : (
   422→                      <span
   423→                        className={`w-1.5 h-1.5 rounded-full ${
   424→                          ch.mute
   425→                            ? 'bg-rose-500'
   426→                            : engineRunning && rmsDb > -52
   427→                              ? 'bg-[#4ADE80] shadow-[0_0_5px_rgba(74,222,128,0.85)]'
   428→                              : 'bg-slate-600'
   429→                        }`}
   430→                      />
   431→                    )}

   433→
   434→                  {/* Column 2: Vertical 10-Segment Live LED Meter */}
   435→                  <div
   436→                    className={`w-2 h-11 rounded-[2px] bg-[#080C10] border p-[1px] flex flex-col justify-between shrink-0 ${
   437→                      !isAssigned ? 'border-[#1F2937]' : 'border-[#2F3E4F]'

   439→                    aria-label={`Channel ${ch.index} Level`}

   441→                    {SEGMENTS_10.slice()

   454→                              : 'bg-[#1D161A]'
   455→                            : segIdx >= 6
   456→                              ? lit
   457→                                ? 'bg-amber-400'
